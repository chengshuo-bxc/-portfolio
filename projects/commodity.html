reverse process

逆向过程

  • 从 price.mysteel.com/assets/request.0efb010a.js 定位签名生成逻辑
  • Api-Token 算法：MD5("path" + "/jgzx" + 业务路径 + "timestamp" + 毫秒 + "version1.0.0" + encrypt_key) 转大写
  • 响应解密机制：isEncrypted=true 时，responseSecretKey / responseKeyOffset 分别是 AES-128-CBC 的密钥和 IV（base64），response 是密文
  • 流程：先用 Node 子进程跑抽取的 JS 验证最快打通，再逐段重写为纯 Python，与浏览器真实请求 cURL 对比匹配
  • 如实记录：SPA 价格接口对非会员返回 price=0（会员门禁），生产数据走 search.ms 公开接口（免登录真实价格）

core/token_crypto.py 
# [逆向要点1] Api-Token 签名: MD5(拼接串) 转大写
# 密钥从前端 JS 提取, 拼法:
# "path" + "/jgzx" + 业务路径 + "timestamp" + 毫秒时间戳 + "version1.0.0" + encrypt_key
MYSTEEL = {
    "app_key": "49B837D39D604919956FCE261BF50060",  # 普通通道 appKey
    "encrypt_key": "d778f81f14774bc0",              # 加密通道签名密钥(逆向自前端)
    "version": "1.0.0",
    "base": "/avgprice/api/jgzx",
}

def mysteel_sign(plain):
    import hashlib
    return hashlib.md5(plain.encode()).hexdigest().upper()

def get_api_token(url, params=None):
    # [逆向要点] 时间戳必须与 token 同源(毫秒), 且一并填进请求参数
    import time
    ts = str(int(time.time() * 1000))
    biz_path = url.replace(MYSTEEL["base"], "").split("?")[0]
    token = mysteel_sign(
        f"path/jgzx{biz_path}timestamp{ts}version{MYSTEEL['version']}{MYSTEEL['encrypt_key']}"
    )
    params["Timestamp"] = ts
    params["Api-Token"] = token
    return token

# [逆向要点2] 响应 value 加密: AES-128-CBC
# 密钥和 IV 直接藏在响应里(base64), response 是密文
def decrypt_value(encrypted_value):
    import base64
    from Crypto.Cipher import AES
    from Crypto.Util.Padding import unpad

    if not encrypted_value.get("isEncrypted"):
        return encrypted_value
    key = base64.b64decode(encrypted_value["responseSecretKey"])  # AES 密钥
    iv  = base64.b64decode(encrypted_value["responseKeyOffset"])  # IV
    cipher = base64.b64decode(encrypted_value["response"])        # 密文
    plain = unpad(AES.new(key, AES.MODE_CBC, iv=iv).decrypt(cipher), 16)
